=====================================

Si un usuario reporta que queries a police_members fallan o se quedan cargando,
ejecuta estos scripts en el SQL Editor de Supabase (https://app.supabase.com)


1. VERIFICACIÓN BÁSICA - ¿La tabla existe y es accesible?

-- Test 1: Ver estructura exacta
SELECT 
  column_name,
  data_type,
  is_nullable,
  column_default
FROM information_schema.columns
WHERE table_name = 'police_members'
ORDER BY ordinal_position;

RESULTADO ESPERADO:
  discord_id | text | NO | NULL
  username | text | YES | NULL
  role_id | text | NO | NULL
  created_at | timestamp with time zone | NO | now()

-- Si resultado > 10000: considerar partición o índices

-- Debe retornar en < 1 segundo

FROM pg_tables

2. DIAGNÓSTICO DE PERFORMANCE - ¿Por qué lenta?
────────────────────────────────────────────────

-- Test 5: Listar todos los índices (mostrará si faltan)
SELECT
  indexname,
  indexdef
FROM pg_indexes
WHERE tablename = 'police_members';

RESULTADO ESPERADO:
  police_members_pkey | CREATE UNIQUE INDEX police_members_pkey ON public.police_members USING btree (discord_id)

Si FALTA índice en role_id o username, ese es el problema.


-- Test 6: Simular query más lenta (búsqueda por role_id)
EXPLAIN ANALYZE
SELECT discord_id, username 
FROM police_members 
WHERE role_id = '1502940172348690482'
LIMIT 100;

BUSCAR EN RESULTADO:
  ✗ "Seq Scan" = búsqueda secuencial (LENTO)
  ✓ "Index Scan" = uso de índice (RÁPIDO)

Si ves Seq Scan, necesitas crear índice:
  CREATE INDEX idx_police_members_role_id ON police_members(role_id);


-- Test 7: Simular búsqueda por username con ILIKE
EXPLAIN ANALYZE
SELECT discord_id, username
FROM police_members
WHERE username ILIKE '%admin%'
LIMIT 50;

Si ves Seq Scan, crear índice:
  CREATE INDEX idx_police_members_username ON police_members(username);


-- Test 8: Buscar locks activos (si tabla está bloqueada)
SELECT 
  l.locktype,
  l.database,
  l.relation::regclass,
  l.mode,
  l.granted,
  a.usename,
  a.query,
  a.state
FROM pg_locks l
JOIN pg_stat_activity a ON l.pid = a.pid
WHERE l.relation = (
  SELECT oid FROM pg_class WHERE relname = 'police_members'
);

Si hay resultados con granted = false: alguien está bloqueando


-- Test 9: Estadísticas de RLS

3. SOLUCIÓN 1: CREAR ÍNDICES FALTANTES
──────────────────────────────────────

-- Índice en role_id (búsqueda frecuente de permisos)
CREATE INDEX IF NOT EXISTS idx_police_members_role_id 
ON police_members(role_id);

-- Índice en username (búsquedas ILIKE)
CREATE INDEX IF NOT EXISTS idx_police_members_username 
ON police_members(username);

-- Validar índices creados
SELECT indexname, indexdef FROM pg_indexes 
WHERE tablename = 'police_members';

5. SOLUCIÓN 3: VERIFICAR POLÍTICAS RLS
──────────────────────────────────────

-- Ver política actual
SELECT 
  policyname,
  permissive,
  roles,
  qual,
  with_check
FROM pg_policies
WHERE schemaname = 'public' AND tablename = 'police_members';

RESULTADO ESPERADO:
  policyname = Allow all on police_members
  permissive = true
  roles = {public}
  qual = true
  with_check = true


-- Si necesitas permitir acceso anón key específicamente:
DROP POLICY IF EXISTS "Allow all on police_members" ON police_members;

CREATE POLICY "Allow all on police_members"
  ON police_members
  FOR ALL
  USING (true)
  WITH CHECK (true);
